\section{Tổng kết}
\subsection*{1. Kỹ năng và kiến thức thu nhận được}
\begin{itemize}
    \item Kiến thức chuyên sâu về thiết kế và triển khai kiến trúc Serverless trên AWS (Lambda, API Gateway, DynamoDB, Amplify).
    \item Kỹ năng quản trị tài khoản, phân quyền (IAM, AWS Organizations), và thiết lập cơ sở dữ liệu NoSQL tối ưu chi phí.
    \item Kỹ năng triển khai ứng dụng web frontend và cấu hình luồng dữ liệu (CORS, Proxy).
    \item Trải nghiệm làm việc nhóm thực tế, quản lý mã nguồn qua Git/GitHub và tự động hóa triển khai bằng CI/CD (GitHub Actions).
\end{itemize}

\subsection*{2. Cảm nhận cá nhân}
\begin{itemize}
    \item \textbf{Về doanh nghiệp (AWS Vietnam):} Môi trường làm việc chuyên nghiệp, năng động. Mentor tận tâm, luôn khuyến khích sinh viên tự tìm tòi và áp dụng các công nghệ Đám mây mới nhất vào bài toán thực tiễn.
    \item \textbf{Về chương trình thực tập:} Chương trình First Cloud Journey vô cùng thực tiễn, giúp sinh viên không chỉ học lý thuyết mà còn tự tay xây dựng một hệ thống hoàn chỉnh từ con số không, tạo nền tảng vững chắc cho con đường Kỹ sư Cloud.
\end{itemize}
